[02] What-If Net Financial Benefit ($) Retention Model 
Campaign Net Financial Benefit ($) = 
VAR TargetedCustomers = 
    CALCULATE(COUNTROWS(Fact_Customer_Churn), Dim_Risk_Tiers[RiskTier] IN {"High Risk", "Medium Risk"})
VAR CampaignCost = TargetedCustomers * SELECTEDVALUE('Param_Cost'[Cost], 120)
VAR GrossSaved = 
    CALCULATE(SUM(Fact_Customer_Churn[ValueAtRisk]), Dim_Risk_Tiers[RiskTier] IN {"High Risk", "Medium Risk"}) * 0.28
RETURN
    GrossSaved - CampaignCost
